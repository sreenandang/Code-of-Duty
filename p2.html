import re
from typing import Dict, Any

class SentinelGuardrail:
    def __init__(self, trust_threshold: float = 0.85):
        self.trust_threshold = trust_threshold
        self.injection_patterns = [
            r"(?i)(ignore previous instructions|system override|export memory)",
            r"(?i)(curl|wget|webhook|eval\(|exec\()",
            r"(?i)(dan\s+mode|do anything now|unrestricted)"
        ]

    def audit_input(self, user_prompt: str) -> Dict[str, Any]:
        for pattern in self.injection_patterns:
            if re.search(pattern, user_prompt):
                return {
                    "verdict": "BLOCKED",
                    "reason": "High-confidence injection payload detected",
                    "safe": False
                }
        return {"verdict": "PASS", "safe": True}
